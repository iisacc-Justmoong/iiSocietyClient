cmake_minimum_required(VERSION 3.24)
project(iiSocietyClient VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core Network)
find_package(iiSocietySync 0.8.0 CONFIG REQUIRED)
find_package(iiAcountManager 0.2.8 CONFIG REQUIRED HINTS "$ENV{HOME}/.local/SDK/iiAcountManager")
set(CMAKE_AUTOMOC ON)
add_library(iiSocietyClient STATIC
    src/iiSocietyClient/Client.h src/iiSocietyClient/Client.cpp src/iiSocietyClient/AccountSession.h src/iiSocietyClient/AccountSession.cpp src/iiSocietyClient/DeviceInfo.h src/iiSocietyClient/DeviceInfo.cpp
    src/iiSocietyClient/DiscoveryService.h src/iiSocietyClient/NearbyDevices.h src/iiSocietyClient/NearbyDevices.cpp src/iiSocietyClient/AutomaticPairing.h src/iiSocietyClient/AutomaticPairing.cpp
    src/iiSocietyClient/PairingCredentials.h src/iiSocietyClient/PairingCredentials.cpp src/iiSocietyClient/HybridDiscoveryService.cpp
    src/iiSocietyClient/GroupSessionStore.h src/iiSocietyClient/GroupSessionStore.cpp src/iiSocietyClient/StateCrypto.h src/iiSocietyClient/StateCrypto.cpp src/iiSocietyClient/iiSocietyClientExport.h)
add_library(iiSocietyClient::iiSocietyClient ALIAS iiSocietyClient)
target_compile_features(iiSocietyClient PUBLIC cxx_std_23)
target_compile_definitions(iiSocietyClient PUBLIC IISOCIETYCLIENT_STATIC PRIVATE SOCIETY_APP_VERSION="0.1.0")
target_include_directories(iiSocietyClient PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>" "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>")
target_link_libraries(iiSocietyClient PUBLIC iiSocietySync::iiSocietySync iiAcountManager::iiAcountManager Qt6::Network)
set_target_properties(iiSocietyClient PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)
if(APPLE)
    find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Concurrent)
    target_link_libraries(iiSocietyClient PRIVATE Qt6::Concurrent)
    enable_language(OBJCXX)
    target_sources(iiSocietyClient PRIVATE src/iiSocietyClient/AppleStateLocation.mm)
    set_source_files_properties(src/iiSocietyClient/AppleStateLocation.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
    target_link_libraries(iiSocietyClient PRIVATE "-framework Foundation" "-framework Security")
    set_target_properties(iiSocietyClient PROPERTIES OBJCXX_STANDARD 23 OBJCXX_STANDARD_REQUIRED ON)
endif()
if(IOS)
    if(NOT CMAKE_GENERATOR STREQUAL "Xcode" AND NOT CMAKE_Swift_COMPILER_TARGET)
        set(CMAKE_Swift_COMPILER_TARGET "arm64-apple-ios${CMAKE_OSX_DEPLOYMENT_TARGET}")
        if(CMAKE_OSX_SYSROOT MATCHES "iphonesimulator")
            string(APPEND CMAKE_Swift_COMPILER_TARGET "-simulator")
        endif()
    endif()
    enable_language(Swift)
    target_sources(iiSocietyClient PRIVATE src/iiSocietyClient/IosDeviceInfo.mm)
    add_library(iiSocietyClientCrypto STATIC src/iiSocietyClient/IosStateCrypto.swift)
    set_target_properties(iiSocietyClientCrypto PROPERTIES AUTOMOC OFF Swift_LANGUAGE_VERSION 5)
    target_compile_options(iiSocietyClientCrypto PRIVATE "$<$<COMPILE_LANGUAGE:Swift>:-parse-as-library>")
    target_link_libraries(iiSocietyClientCrypto PRIVATE "-framework CryptoKit" "-framework Foundation")
    target_link_libraries(iiSocietyClient PRIVATE iiSocietyClientCrypto "-framework UIKit")
    install(TARGETS iiSocietyClientCrypto EXPORT iiSocietyClientTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
elseif(ANDROID)
    set(SOCIETY_ANDROID_OPENSSL_DIR "" CACHE PATH "OpenSSL runtime directory")
    target_include_directories(iiSocietyClient PRIVATE "${SOCIETY_ANDROID_OPENSSL_DIR}/../include")
    target_link_libraries(iiSocietyClient PRIVATE "${SOCIETY_ANDROID_OPENSSL_DIR}/libcrypto_3.so")
else()
    find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
    target_link_libraries(iiSocietyClient PRIVATE OpenSSL::Crypto)
endif()
if(ANDROID)
    target_sources(iiSocietyClient PRIVATE src/iiSocietyClient/AndroidDiscoveryService.cpp)
else()
    find_path(IISOCIETYCLIENT_DNSSD_INCLUDE dns_sd.h PATH_SUFFIXES avahi-compat-libdns_sd REQUIRED)
    target_sources(iiSocietyClient PRIVATE src/iiSocietyClient/DnsSdDiscoveryService.cpp)
    target_include_directories(iiSocietyClient PRIVATE "${IISOCIETYCLIENT_DNSSD_INCLUDE}")
    if(NOT APPLE)
        find_library(IISOCIETYCLIENT_DNSSD_LIBRARY NAMES dns_sd dnssd REQUIRED)
        target_link_libraries(iiSocietyClient PRIVATE "${IISOCIETYCLIENT_DNSSD_LIBRARY}")
    endif()
endif()
install(TARGETS iiSocietyClient EXPORT iiSocietyClientTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES src/iiSocietyClient/Client.h src/iiSocietyClient/AccountSession.h src/iiSocietyClient/DeviceInfo.h src/iiSocietyClient/DiscoveryService.h src/iiSocietyClient/NearbyDevices.h src/iiSocietyClient/AutomaticPairing.h
    src/iiSocietyClient/PairingCredentials.h src/iiSocietyClient/GroupSessionStore.h src/iiSocietyClient/StateCrypto.h src/iiSocietyClient/iiSocietyClientExport.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/iiSocietyClient)
install(FILES README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/iiSocietyClient)
configure_package_config_file(cmake/iiSocietyClientConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/iiSocietyClientConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiSocietyClient)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/iiSocietyClientConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT iiSocietyClientTargets NAMESPACE iiSocietyClient:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiSocietyClient)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/iiSocietyClientConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/iiSocietyClientConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iiSocietyClient)

if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Test)
    add_executable(iiSocietyClientTests tests/client.cpp)
    target_link_libraries(iiSocietyClientTests PRIVATE iiSocietyClient Qt6::Test)
    target_compile_definitions(iiSocietyClientTests PRIVATE SOCIETY_ACCOUNT_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/tests/account.json" CLIENT_TEST_DIRECTORY="${CMAKE_CURRENT_BINARY_DIR}")
    add_test(NAME iiSocietyClient.Session COMMAND iiSocietyClientTests)
    set_tests_properties(iiSocietyClient.Session PROPERTIES TIMEOUT 60 ENVIRONMENT "SOCIETY_DISABLE_SESSION_RESTORE=1")
    if(APPLE AND NOT IOS)
        add_executable(iiSocietyClientAppleStateTests tests/apple_state.mm)
        target_link_libraries(iiSocietyClientAppleStateTests PRIVATE iiSocietyClient Qt6::Test "-framework Security")
        set_target_properties(iiSocietyClientAppleStateTests PROPERTIES OBJCXX_STANDARD 23)
        add_test(NAME iiSocietyClient.AppleState COMMAND iiSocietyClientAppleStateTests)
    endif()
endif()
